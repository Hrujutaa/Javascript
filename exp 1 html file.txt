  <!-- INTERNAL JS: inside a script tag in this file -->
  <script>
    console.info('Internal JS: page script loaded');

    document.getElementById('welcome').textContent = 'Welcome to the JavaScript Demo Page!';

    function showUser() {
      const name = document.getElementById('name').value.trim();
      const age = document.getElementById('age').value.trim();
      const error = document.getElementById('error');
      error.textContent = '';

      if (name === '' || age === '') {
        console.error('Internal JS: name or age is missing');
        error.textContent = 'Please enter both name and age.';
        return;
      }

      if (Number(age) < 1 || Number(age) > 120) {
        console.warn('Internal JS: unusual age entered:', age);
      }

      console.table({ name, age });
      document.getElementById('userInfo').innerHTML =
        `Name: ${name}<br>Age: ${age}<br>Welcome, ${name}!`;
    }
  </script>
